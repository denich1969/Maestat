! One pass over the NPCs of a world: args[0] = 1..5 for a city district, otherwise the players world.
! args[1]=1: the players world - NPCs walk step by step and the player sees them come and go.
! args[1]=0: a background district - NPCs appear straight at the place they are heading to.
! An NPC that has nothing to do until its wake time costs only the few checks below.
ns_pk=args[0]
ns_act=args[1]
killvar 'nsc'
killvar 'nsm'
killvar 'nsm_n'
killvar 'nsm_busy'
if ns_pk=1:copyarr 'nsc','nsb1'
if ns_pk=2:copyarr 'nsc','nsb2'
if ns_pk=3:copyarr 'nsc','nsb3'
if ns_pk=4:copyarr 'nsc','nsb4'
if ns_pk=5:copyarr 'nsc','nsb5'
if ns_pk<1 or ns_pk>5:copyarr 'nsc','nsbx'
ns_obs=0
if ns_pk=700 and ns_act=1:ns_obs=1
ns_cn=arrsize('nsc')
ns_ci=0
:ns_passloop
if ns_ci<ns_cn:
	w=nsc[ns_ci]
	ns_w=w
	if ns_obs=1:gs'npcsys_obs' & w=ns_w
	ns_ix=0
	ns_c_pass+=1
	if w<>you and dead[w]=0:
		ns_ix=1
		! slow bookkeeping: stats, clothes, relationships
		if nsu_next[w]<=time:gs'npcsys_upkeep' & w=ns_w
		! followers stay with the player
		if follower[w]=you:X[w]=X[you] & Y[w]=Y[you] & Z[w]=Z[you] & H[w]=H[you] & R[w]=R[you]
		! NPCs made without an age get one as soon as they can walk (as the old walker did every turn)
		if age[w]<=0 and cantwalk[w]=0:age[w]=iif(mob[w]>0,rand(18,30),rand(18,25))
		! a finished activity (sex, a fight...) frees the NPC
		if mob_busy[w]>0 and mob_busy[w]<time:
			mob_sleep[w]=time+iif(mob[w]>0 and mob_busy[w]=1,60,15)
			horny[w]=0 & mob_busy[w]=0 & under_sub[w]=0 & action_tipe[w]=0
		end
		if cantwalk[w]=0 and follower[w]=0 and immobilaze[w]=0 and age[w]>0 and mob_busy[w]=0 and walker_cooldown[w]<time:
			ns_c_ok+=1
			if cum_face[w]>0:
				cum_face[w]=0
				if ns_act=1 and X[w]=X[you] and Y[w]=Y[you] and Z[w]=Z[you] and H[w]=H[you]:gs'npcsys_say','вытирает сперму' & w=ns_w
			end
			if nsw_wake[w]<=time:gs'npcsys_decide' & w=ns_w
			if nsw_mode[w]>0 or X[w]<>Xdest[w] or Y[w]<>Ydest[w] or Z[w]<>Zdest[w] or H[w]<>Hdest[w] or R[w]<>Rdest[w]:gs'npcsys_move' & w=ns_w
		elseif ns_dbg>0:
			gs'npcsys_dbg_skip'
		end
	end
	! bodies in the obsession world can still be raised by the shaman
	if ns_obs=1 and w<>you and (dead[w]=1 or dead[w]=3):ns_ix=1
	! note who stands where, for npcsys_social
	if ns_ix=1:
		$ns_k='<<X[w]>>,<<Y[w]>>,<<Z[w]>>,<<H[w]>>,<<R[w]>>'
		ns_kc=nsm_n[$ns_k]
		nsm['<<$ns_k>>#<<ns_kc>>']=w
		nsm_n[$ns_k]=ns_kc+1
		if ns_kc=1:$nsm_busy[]=$ns_k
	end
end
ns_ci+=1
if ns_ci<ns_cn:jump'ns_passloop'
