	end

if ns_dn>0:$ns_rep+='НПС с неописанной программой (стоят на месте), программа - сколько НПС:'

:ns_dploop
if ns_di<ns_dn:$ns_rep+=' <<ns_dpl[ns_di]>> - <<ns_dup[ns_dpl[ns_di]]>>;'
ns_di+=1
if ns_di<ns_dn:jump'ns_dploop'

killvar 'ns_dup' & killvar 'ns_dpl'

$ns_rep+='<br><b>старые списки</b> (размер/прочитано):<br>'
gs'npcsys_dbg_list','homeworld'
gs'npcsys_dbg_list','paradiseworld'

gs'npcsys_dbg_list','se_world'

gs'npcsys_dbg_list','romaworld'
gs'npcsys_dbg_list','mistworld'

gs'npcsys_dbg_list','l_obsession_world'
gs'npcsys_dbg_list','t_obsession_world'
gs'npcsys_dbg_list','m_obsession_world'

			gs'npcsys_dbg_npc',ns_dw

		end

gs'npcsys_dbg_scan','n_world'
