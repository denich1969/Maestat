		if X[w]=nsl_x['кладбище'] and Y[w]=nsl_y['кладбище'] and Z[w]=nsl_z['кладбище'] and H[w]=nsl_h['кладбище']:kelengrave+=1 & bossQW=2 & item_loc['82,<<$LOC_str[w]>>']+=1

		end
